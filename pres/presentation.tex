\begin{frame}{В предыдущих сериях}
    \begin{itemize}
        \item<1-> \(\lambda(t)\) -- кусочно-постоянный процесс с \(n\) состояниями: \(S_1,S_2,\dots,S_n\);
        \item<2-> \(\lambda(t)\) -- принципиально ненаблюдаемый случайный марковский процесс;
        \item<3-> Если \(\lambda(t)=\lambda_i\), то имеет место \(S_i\)-е состояние процесса;
        \item<4-> \(\lambda(t),i=\overline{1,n}\);
        \item<5-> \(\lambda_1>\lambda_2>\dots>\lambda_n\);
        \item<6-> \(\eta=\min(\eta^{(1)},\eta^{(2)})\);
        \item<7-> \(\eta^{(1)}:F_i^{(1)}(t)=1-e^{-\lambda_it}, \quad \eta^{(2)}:F_i^{(2)}(t)=1-e^{-\alpha_it}\);
        \item<8-> По методу обратных функций: \(\eta^{(1)}=-\frac{1}{\lambda_i}ln(1-x_1), \quad \eta^{(2)}=-\frac{1}{\alpha_i}ln(1-x_2)\), где \(x_1,x_2\in[0;1)\);
        \item<9-> \(P_1^{(1)}(\lambda_j|\lambda_i), \quad P_1^{(2)}(\lambda_j|\lambda_i), \quad i,j=\overline{1,n}\);
        \item<10-> \(\sum_{i=1}^n P_1^{(1)}(\lambda_j|\lambda_i)=1,j=\overline{1,n}; \quad \sum_{i=1}^n P_1^{(2)}(\lambda_j|\lambda_i)=1, j=\overline{1,n}\)
    \end{itemize}
\end{frame}

\begin{frame}{Введение}
    \textbf{Цель} -- уметь по наблюдениям за моментами времени \(t_1,t_2,\dots,\) наступления событий обобщенного синхронного потока второго порядка с произвольным числом состояний на интервале \((t_0,t)\) оценить состояние процесса \(\lambda(t)\) в момент времени \(t\).
    \newline

    Для этого необходимо знать явный вид апостериорных вероятностей \(w(\lambda_i|t_1,t_2,\dots,t_m,t)=P(\lambda(t)=\lambda_i|t_1,t_2,\dots,t_m,t)=w(\lambda_i|t)\)
    \newline

    Вывод о том, какое состояние процесса \(\lambda(t)\) имеет место в момент времени \(t\), выносится по критерию максимума апостериорной вероятности:
    \[w(\lambda_i|t)\geq w(\lambda_j|t), j\ne i,\quad i,j=\overline{1,n} \Rightarrow \hat{\lambda}(t)=\lambda_i\]
\end{frame}

\begin{frame}{Введение}
    Задачи, которые необходимо сделать для достижения поставленной цели:
    \begin{enumerate}
        \item Вывести инфинитезимальные характеристики потока.
        \item Вывести рекуррентное соотношение для апостериорных вероятностей для дважды стохастических потоков событий с произвольным числом состояний.
        \item Преобразовать эти рекуррентные соотношения, рассмотрев два случая.
        \item Вывести формулы для априорных финальных вероятностей состояний процесса \(\lambda(t)\).
    \end{enumerate}
\end{frame}

\begin{frame}{Инфинитезимальные характеристики потока}
    \[D_0=
    \begin{bmatrix}
        -(\lambda_1+\alpha_1) & 0 & \dots & 0 \\
        0 & -(\lambda_2+\alpha_2) & \dots & 0 \\
        \dots & \dots & \dots & \dots \\
        0 & 0 & \dots & -(\lambda_n+\alpha_n)
    \end{bmatrix}
    \]
    {\color{black}
\small
\setlength{\arraycolsep}{2pt}
\[D_1=
    \begin{bmatrix}
        \lambda_1P_1^{(1)}(\lambda_1|\lambda_1)+\alpha_1P_1^{(2)}(\lambda_1|\lambda_1) & \dots & \lambda_1P_1^{(1)}(\lambda_n|\lambda_1)+\alpha_1P_1^{(2)}(\lambda_n|\lambda_1) \\
        \lambda_2P_1^{(1)}(\lambda_1|\lambda_2)+\alpha_2P_1^{(2)}(\lambda_1|\lambda_2) & \dots & \lambda_2P_1^{(1)}(\lambda_n|\lambda_2)+\alpha_2P_1^{(2)}(\lambda_n|\lambda_2) \\
        \dots & \dots &  \dots \\
        \lambda_nP_1^{(1)}(\lambda_1|\lambda_n)+\alpha_nP_1^{(2)}(\lambda_1|\lambda_n) & \dots & \lambda_nP_1^{(1)}(\lambda_n|\lambda_n)+\alpha_nP_1^{(2)}(\lambda_n|\lambda_n)
    \end{bmatrix}
\]
}
\end{frame}

\begin{frame}{Математическая модель}
    Пусть наблюдение за потоком начинается в момент времени \(t=0\); время \(t\) меняется дискретно с шагом \(\Delta t:t^{(l)}=l\Delta t, l=0,1,\dots \). Введем двумерный случайный процесс \((\lambda^{(l)},r_l)\), где \(\lambda^{(l)}=\lambda(l\Delta t)\) -- значение процесса в момент времени \(t^{(l)}=l\Delta t \quad (\lambda^{(l)}=\lambda_i, i=\overline{1,n})\);\(\quad r_l=r_l(\Delta t)=r(l\Delta t) - r((l-1)\Delta t)\) -- число событий потока, наступивших на интервале времени \(((l-1)\Delta t,l\Delta t)\) длительности \(\Delta t\). 
    \newline

    \(r_0\) можно положить равным нулю, так как на интервале \((-\Delta t,0)\) наблюдение за потоком не производится.
\end{frame}

\begin{frame}
    \begin{rlemma}[1]
        Для обобщенного синхронного потока событий второго порядка с произвольным числом состояний случайный процесс \((\lambda^{(k)},r_k)\) является марковским
    \end{rlemma}
\end{frame}

\begin{frame}{Математическая модель}
    Для реализации алгоритма оптимального оценивания состояния процесса \(\lambda(t)\) по наблюдениям за потоком событий требуется найти апостериорную вероятность \(w(\lambda_i|\boldsymbol{r}_m)\) того, что в момент времени \(m\Delta t\) процесс \(\lambda(t)\) находится в состоянии \(\lambda_i\), т.е. \(\lambda(m\Delta t)=\lambda_i, i=\overline{1,n}\). Здесь \(w(\lambda_i|\boldsymbol{r}_m)=1-\sum_{j=1}^{n}w(\lambda_i|\boldsymbol{r}_m),i\ne j\). 
    \newline

    Обозначим \(p(\lambda^{(l)},r_l|\lambda^{(l-1)},r_{l-1})\) -- вероятность перехода процесса \((\lambda^{(l)},r_l)\) за один шаг \(\Delta t\) из \((\lambda^{(l-1)},r_{l-1})\) в состояние \((\lambda^{(l)},r_l)\); \(w(\boldsymbol{\lambda}^{(m)}|\boldsymbol{r}_m)\) -- совместная вероятность значений дискретных случайных величин \(\boldsymbol{\Lambda}^{(m)},\boldsymbol{R}^{(m)}\). Здесь \(\boldsymbol{\Lambda}^{(m)}=(\Lambda^{(0)},\dots,\Lambda^{(m)})\) -- вектор случайных величин, \(\boldsymbol{\lambda}^{(m)}=(\lambda^{(0)},\dots,\lambda^{(m)})\) -- вектор значений случайных величин \(\boldsymbol{\Lambda}^{(m)}\); \(\boldsymbol{R}^{(m)}=(R^{(0)},\dots,R^{(m)})\) -- вектор случайных величин,\(\boldsymbol{r}_m=(r_0,\dots,r_m)\) -- вектор значений случайной величины \(\boldsymbol{R}^{(m)}\). 
\end{frame}
